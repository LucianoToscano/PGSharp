name: Backup de fonte PGSharp-Privado para Drive
on:
  push:
    paths:
      - '.github/workflows/backup-source-for-drive.yml'
  workflow_dispatch:
permissions:
  contents: read
jobs:
  backup:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - name: Empacotar fonte versionada e SHA-256
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p dist
          git archive --format=zip --output=dist/PGSharp-Privado-FONTE-COMPLETA.zip HEAD
          git bundle create dist/PGSharp-Privado-HISTORICO.bundle --all
          git bundle verify dist/PGSharp-Privado-HISTORICO.bundle
          sha256sum dist/PGSharp-Privado-FONTE-COMPLETA.zip dist/PGSharp-Privado-HISTORICO.bundle > dist/SHA256SUMS.txt
      - uses: actions/upload-artifact@v4
        with:
          name: PGSharp-Privado-FONTE-COMPLETA-PARA-DRIVE
          path: dist/
          if-no-files-found: error
          retention-days: 90
